\section{Compact TT propagation and the closed theorem}
\label{sec:ttcombined}

\subsection{Normalized compact quadrupole channel}

For an STF quadrupole amplitude $Q_{ij}$ and direction $\hat{\bm n}$, let
\begin{equation}
\Lambda_{ij,kl}(\hat{\bm n})
=\frac12(P_{ik}P_{jl}+P_{il}P_{jk}-P_{ij}P_{kl}),
\qquad P_{ij}=\delta_{ij}-n_in_j,
\label{eq:TTprojector}
\end{equation}
be the TT projector. Rotational invariance gives
\begin{equation}
\int\dd\Omega\,\Lambda(\hat{\bm n})
=\frac{8\pi}{5}I_{\rm STF},
\label{eq:LambdaIntegral}
\end{equation}
so the normalized directivity is
\begin{equation}
D_Q(\hat{\bm n})
=\frac52\frac{Q^*:\Lambda(\hat{\bm n}):Q}{Q^*:Q}
\le\frac52.
\label{eq:Dmax}
\end{equation}
The functional and its $5/2$ maximum are historical compact gravitational-antenna physics \cite{HirakawaNariharaFujimoto1976}.

For direct normalization of the propagation map, define the unit angular radiation mode
\begin{equation} u_{Q,\lambda}(\hat{\bm n})
=\sqrt{\frac5{8\pi}}
\frac{Q:\epsilon^{(\lambda)}(\hat{\bm n})}{\sqrt{Q^*:Q}},
\label{eq:normalizedmode}
\end{equation}
with $\sum_\lambda\int\dd\Omega\,|u_{Q,\lambda}|^2=1$. For source and receiver separated by $R\hat{\bm R}$, the outgoing stationary-phase contribution of the translated overlap is
\begin{equation}
\boxed{
t_{BA}^{\rm TT}
=-\frac{5i}{4kR}\ee^{ikR}
\frac{Q_B^*:\Lambda(\hat{\bm R}):Q_A}
{\sqrt{Q_A^*:Q_A}\sqrt{Q_B^*:Q_B}}
+O((kR)^{-2}).}
\label{eq:TTamplitude}
\end{equation}
Therefore
\begin{equation}
\boxed{
\|P_g(\omega)\|_{\op}^2
\le
\frac{25}{16[k(\omega)R]^2}
\left[1+O((kR)^{-1})\right].}
\label{eq:TTpropbound}
\end{equation}
Matched tensors in the same line-of-sight TT subspace saturate the leading projector inequality. Appendix~\ref{app:TTcheck} gives an independent aligned-plus integral. The coefficient also equals the reciprocal far-field factor
\begin{equation}
D_AD_B\left(\frac{\lambda}{4\pi R}\right)^2
=\frac{25}{16(kR)^2}
\label{eq:FriisTT}
\end{equation}
for $D_A=D_B=5/2$, confirming that the TT normalization and classical antenna interpretation agree.

\subsection{Passive repeated returns}

Repeated scattering between the same two passive separated endpoints does not increase the leading ceiling. Let $R_A,R_B$ be their exact gravitational reflection blocks and $P_{BA},P_{AB}$ the forward/reverse propagation maps. Standard scattering-network composition gives \cite{Redheffer1962}
\begin{equation}
P_{\rm eff}
=(I-P_{BA}R_AP_{AB}R_B)^{-1}P_{BA}.
\label{eq:Peff}
\end{equation}
With $p_+=\|P_{BA}\|_{\op}$ and $p_-=\|P_{AB}\|_{\op}$, passivity implies
\begin{equation}
\boxed{
\|P_{\rm eff}\|_{\op}
\le\frac{p_+}{1-p_+p_-}.}
\label{eq:Peffbound}
\end{equation}
For reciprocal propagation, $p_+=p_-=p$ and $\eta=p^2$, hence
\begin{equation}
\boxed{
\eta_{\rm rec}\le\frac{\eta}{(1-\eta)^2}.}
\label{eq:etarec}
\end{equation}
Since $\eta=O((kR)^{-2})$,
\begin{equation}
\boxed{
\eta_{\rm rec}\le \eta+O((kR)^{-4}).}
\label{eq:recurrentorder}
\end{equation}
Thus the recurrent-scattering \emph{upper bound} differs from the one-hop ceiling only beyond the retained $O((kR)^{-2})$ power term. The actual recurrent transfer may be smaller because of interference. Added relays, mirrors, external cavities, near-field exchange, and nonseparable common interaction regions are different architectures.

\subsection{Inertia-controlled throughput}

Assume a narrow operating band centered at $\omega$ with
\begin{equation}
\frac{\omega L_A}{c}\ll1,
\qquad
\frac{\omega L_B}{c}\ll1,
\qquad
kR=\frac{\omega R}{c}\gg1.
\label{eq:regime}
\end{equation}
Combining Eqs.~\eqref{eq:materialcutset} and \eqref{eq:TTpropbound} gives
\begin{theorem}[Passive compact-quadrupole gravitational throughput]
\begin{equation}
\boxed{
\Gamma_{\rm coh}
\lesssim
\frac{25}{16(kR)^2}
\frac{4G\omega^4}{3c^5}
\min(I_A,I_B).}
\label{eq:combined1}
\end{equation}
Equivalently,
\begin{equation}
\boxed{
\Gamma_{\rm coh}
\lesssim
\frac{25G\omega^2}{12c^3R^2}
\min(I_A,I_B).}
\label{eq:combined2}
\end{equation}
\end{theorem}

The theorem is an upper bound, not a global saturability claim. Its physical content is that the leading integrated ceiling contains no endpoint quality factor, passive resonance count, coherent mixing parameter, or compact orientation. Equations~\eqref{eq:etarec}--\eqref{eq:recurrentorder} add that arbitrary passive returns between the same two separated endpoints cannot increase the leading coefficient.

For a finite band $\mathcal B=[\omega_-,\omega_+]$ entirely inside the compact wave-zone regime, the conservative edge-separated form is
\begin{equation}
\Gamma_{\rm coh}(\mathcal B)
\lesssim
\frac{25G}{12c^3R^2}
\frac{\omega_+^4}{\omega_-^2}
\min(I_A,I_B).
\label{eq:broadband}
\end{equation}
